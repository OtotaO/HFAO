  kubectl port-forward svc/{{ include "hfao.fullname" . }}-cockpit \
      {{ .Values.cockpit.service.httpPort }}:{{ .Values.cockpit.service.httpPort }} \
      {{ .Values.cockpit.service.mcpPort }}:{{ .Values.cockpit.service.mcpPort }}
  open http://localhost:{{ .Values.cockpit.service.httpPort }}            # cockpit
  open http://localhost:{{ .Values.cockpit.service.mcpPort }}/mcp         # MCP

  kubectl port-forward svc/{{ include "hfao.fullname" . }}-ingest \
      {{ .Values.ingest.service.port }}:{{ .Values.ingest.service.port }}

Then point any OTLP-emitting agent at http://localhost:{{ .Values.ingest.service.port }}/v1/traces.
